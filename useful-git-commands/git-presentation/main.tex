\section{The Problem}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{Has your folder ever looked like this?}
    \begin{columns}
        \column{0.5\textwidth}
            \begin{lstlisting}
analysis_v1.py
analysis_v2.py
analysis_v2_fixed.py
analysis_final.py
analysis_final_REAL.py
analysis_final_REAL_submitted.py
analysis_final_REAL_submitted_2.py
            \end{lstlisting}
        \column{0.5\textwidth}
            \pause
            \begin{block}{The hard questions $\triangleright$}
                \begin{itemize}
                    \item Which one did I submit?
                    \item What changed between v2 and final?
                    \item Which one runs on the cluster?
                    \item What did I change last Tuesday?
                \end{itemize}
            \end{block}
    \end{columns}
    \vspace{1em}
    \pause
    \centering
    \textbf{If this looks familiar --- you were already doing version control.}\\[4pt]
    \hspace{0.5\textwidth}\textbf{Just doing it badly.}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{You invented version control (without knowing it)}
    \begin{columns}
        \column{0.48\textwidth}
            \begin{block}{What you do manually $\triangleright$}
                \begin{tabular}{ll}
                    \texttt{\_v1}, \texttt{\_v2}      & $\leftarrow$ version tag    \\[4pt]
                    \texttt{\_fixed}                   & $\leftarrow$ commit message \\[4pt]
                    \texttt{\_final\_REAL}             & $\leftarrow$ release tag    \\[4pt]
                    keeping old files                  & $\leftarrow$ history        \\
                \end{tabular}
            \end{block}
        \column{0.48\textwidth}
            \pause
            \begin{block}{What git does automatically $\triangleright$}
                \begin{tabular}{ll}
                    version number     & assigned           \\[4pt]
                    commit message     & you write it once  \\[4pt]
                    tags               & natively supported \\[4pt]
                    full history       & always, forever    \\
                \end{tabular}
            \end{block}
    \end{columns}
    \vspace{1.2em}
    \pause
    \centering
    \textbf{Git does exactly what you were doing ---\\
    without cluttering your folder.}\\[6pt]
    \pause
    \textit{And with added functionality to make our life much, much easier.}
\end{frame}

    \centering
    \vspace{0.5em}

    \vspace{1em}

\section{What is git}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{git is just a folder that remembers}
    \begin{columns}
        \column{0.52\textwidth}
            \begin{lstlisting}
my_project/
|-- .git/       <- git lives HERE
|-- analysis.py <- tracked by git
|-- results.csv <- NOT tracked by git
`-- README.md   <- tracked by git
            \end{lstlisting}
            \vspace{0.5em}
            \begin{itemize}
                \item No internet required
                \item No account required
                \item A hidden folder inside your project
                \item Everything git knows lives right there
            \end{itemize}
        \column{0.44\textwidth}
    \end{columns}
    \pause
    \begin{tikzpicture}[remember picture, overlay]
        \node[draw=red, line width=2pt, fill=white, rounded corners=4pt,
              text width=0.55\textwidth, align=center, inner sep=14pt]
              at (current page.center) {
            \textbf{\large That is all git is}\\[8pt]
            A tool installed on your computer.\\[6pt]
            It tracks changes to files you tell it to track.\\[6pt]
            It works completely offline.\\[6pt]
            \textit{The moment you run \texttt{git init},\\
            your directory becomes a repository.}
        };
    \end{tikzpicture}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Two different things. One name.}
    \begin{columns}
        \column{0.48\textwidth}
            \begin{block}{\texttt{git} --- the tool $\triangleright$}
                \begin{itemize}
                    \item Runs on your computer
                    \item Works offline
                    \item Free and open source
                    \item No account needed
                    \item History lives in \texttt{.git/}
                \end{itemize}
            \end{block}
        \column{0.48\textwidth}
            \begin{block}{GitHub --- the website $\triangleright$}
                \begin{itemize}
                    \item A website in the cloud
                    \item Stores a \textit{copy} of your repo
                    \item Adds a web interface
                    \item Adds collaboration tools
                    \item Optional --- but very useful
                \end{itemize}
            \end{block}
    \end{columns}
    \vspace{1em}
    \pause
    \centering
    \texttt{[Your laptop \;\; .git/]}
    \quad $\overset{\texttt{push}}{\longrightarrow}$ \quad
    \texttt{[GitHub]}\\[6pt]
    \small
    You need \texttt{git} to use GitHub.\\
    You do \textbf{not} need GitHub to use \texttt{git}.
\end{frame}

\section{Local git}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Local git --- three commands, no internet}
\centering
\vspace{0.5em}
\begin{tikzpicture}[
    gitbox/.style={draw=black, line width=1.2pt, fill=white,
                   inner sep=6pt, text width=4.0cm, align=left, font=\small},
    arr/.style={->, red, line width=0.8pt, shorten >=2pt, shorten <=2pt}
]
    \node[gitbox] (init) {
        \textbf{\texttt{git init}}\\[3pt]
        Start tracking this directory.\\[3pt]
        \texttt{\$ git init}\\[3pt]
        \textit{\scriptsize Creates \texttt{.git/}. Done once per project.}
    };
    \node[gitbox, visible on=<2->, right=0.5cm of init] (add) {
        \textbf{\texttt{git add}}\\[3pt]
        Choose files for the next snapshot.\\[3pt]
        \texttt{\$ git add analysis.py README.md}\\[3pt]
        \textit{\scriptsize Offline. Nothing moves.}
    };
    \node[gitbox, visible on=<3->, right=0.5cm of add] (commit) {
        \textbf{\texttt{git commit}}\\[3pt]
        Save snapshot with a message.\\[3pt]
        \texttt{\$ git commit -m "new plot"}\\[3pt]
        \textit{\scriptsize Still offline. Stored in \texttt{.git/}.}
    };
    \draw[arr, visible on=<2->] (init.east)   -- (add.west);
    \draw[arr, visible on=<3->] (add.east)    -- (commit.west);
\end{tikzpicture}\\[0.8em]
\uncover<4->{\small These three commands are all you need to get started.\\[4pt]
\textbf{None of them touch the network.}}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{git status --- what git sees right now}
    \begin{columns}
        \column{0.44\textwidth}
            \uncover<2->{
            \begin{block}{Three zones $\triangleright$}
                \begin{itemize}
                    \item \textbf{Staged} --- ready to commit
                    \item \textbf{Modified} --- changed, not staged yet
                    \item \textbf{Untracked} --- git does not know this file
                \end{itemize}
            \end{block}
            }
            \vspace{0.4em}
            \uncover<3->{
            \begin{block}{Moving between zones $\triangleright$}
                Stage a file:\\
                \texttt{git add README.md}\\[4pt]
                Changed your mind? Unstage it:\\
                \texttt{git restore --staged README.md}
            \end{block}
            }
        \column{0.52\textwidth}
            \begin{lstlisting}
$ git status
On branch main

Changes to be committed:
    modified: analysis.py

Changes not staged for commit:
    modified: README.md

Untracked files:
    results.csv
            \end{lstlisting}
            \vspace{0.4em}
            \uncover<4->{
            \begin{tcolorbox}[colback=red!10, colframe=red, boxrule=1pt,
                              arc=2pt, boxsep=4pt]
                \centering\textit{Run \texttt{git status} before anything else. Always.}
            \end{tcolorbox}
            }
    \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{git diff --- what exactly changed}
    \begin{columns}
        \column{0.44\textwidth}
            \uncover<2->{
            \begin{block}{Reading the output $\triangleright$}
                \begin{itemize}
                    \item Lines with \texttt{-} were removed
                    \item Lines with \texttt{+} were added
                    \item Unchanged lines have no prefix
                \end{itemize}
            \end{block}
            }
            \vspace{0.4em}
            \uncover<3->{
            \begin{block}{Useful variants $\triangleright$}
                \texttt{git diff} \quad since last commit\\[4pt]
                \texttt{git diff main} \quad vs main branch
            \end{block}
            }
            \vspace{0.4em}
            \uncover<4->{
            \begin{tcolorbox}[colback=red!10, colframe=red, boxrule=1pt,
                              arc=2pt, boxsep=4pt]
                \centering\textit{More useful than opening two files and comparing manually.}
            \end{tcolorbox}
            }
        \column{0.52\textwidth}
            \begin{lstlisting}
$ git diff README.md
-## Old section heading
+## New section heading
 Some line that did not change
-This line was removed
+This line was added
            \end{lstlisting}
    \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{git log --- your full history}
    \begin{columns}
        \column{0.44\textwidth}
            \uncover<2->{
            \begin{block}{Useful variants $\triangleright$}
                \begin{itemize}
                    \item \texttt{git log} --- full commit history
                    \item \texttt{git log -p} --- history with changes
                    \item \texttt{git log -p file.py} --- history of one file
                \end{itemize}
            \end{block}
            }
            \vspace{0.4em}
            \uncover<3->{
            \begin{tcolorbox}[colback=red!10, colframe=red, boxrule=1pt,

                \centering\textit{Every commit is signed with your name, email.\\
                Your work diary, automatic.}

            \begin{lstlisting}
$ git log

Author: Suryanarayan Mondal
Date:   Mon Apr 13 2026

    added energy calibration

commit 7b1e84a
Author: Suryanarayan Mondal

\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{git reflog \& git reset --- your safety net}
    \begin{columns}
        \column{0.44\textwidth}
            \uncover<2->{
            \begin{block}{\texttt{git reflog} $\triangleright$}
                Shows every position \texttt{HEAD} has been.\\[4pt]
                If you ever think you lost work ---\\
                look here first.
            \end{block}
            }
            \vspace{0.4em}
            \uncover<3->{
            \begin{block}{Go back to any previous state $\triangleright$}
                \texttt{git reset --hard 7b1e84a}
            \end{block}
            }
            \vspace{0.4em}
        \column{0.52\textwidth}
            \begin{lstlisting}
$ git reflog
a3f9c2d HEAD@{0}: commit: energy calib
7b1e84a HEAD@{1}: commit: first version
c4d2f1e HEAD@{2}: commit: initial
            \end{lstlisting}
    \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{Branches --- isolated copies of your work}
    \begin{columns}
        \column{0.52\textwidth}
            \uncover<2->{
            \begin{block}{What is a branch? $\triangleright$}
                Remember \texttt{analysis\_v2\_experimental.py}?\\
                A branch does the same thing ---\\
                without creating extra files.\\[4pt]
                Your \texttt{main} branch is always untouched.
            \end{block}
            }
            \vspace{0.4em}
            \uncover<3->{
            \begin{block}{Basic commands $\triangleright$}
                \begin{tabular}{l@{\quad$\triangleright$\quad}l}
                    \texttt{git branch myidea}    & create       \\[4pt]
                    \texttt{git checkout myidea}  & switch to it \\[4pt]
                    \texttt{git checkout main}    & go back      \\[4pt]
                    \texttt{git branch -d myidea} & delete       \\[4pt]
                    \texttt{git merge main}       & merge branches
                \end{tabular}
            \end{block}
            }
        \column{0.4\textwidth}
            \begin{lstlisting}[language=bash]
# create and switch to a new branch
git branch myidea
git checkout myidea

# work, add, commit as usual
git add analysis.py
git commit -m "new approach"

# not happy? go back to main
git checkout main

# delete the branch
git branch -d myidea
            \end{lstlisting}
    \end{columns}
\end{frame}

\section{What git gives you}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{What git gives YOU --- not the team}
    \begin{columns}
        \column{0.48\textwidth}
            \begin{block}{\emph{Ownership with a timestamp} $\triangleright$}
                Every commit is signed, dated,\\
                and attributed to you.\\[4pt]
                \textit{Your work diary --- automatic.}
            \end{block}
            \vspace{0.5em}
            \begin{block}{\emph{A personal time machine} $\triangleright$}
                Revert anything, anytime.\\
                Experiment without fear.\\[4pt]
                \textit{Nothing is ever truly deleted.}
            \end{block}
        \column{0.48\textwidth}
            \begin{block}{\emph{Work from anywhere} $\triangleright$}
                Office, home, cluster.\\
                One sync command and you are up to date.\\[4pt]
                \textit{No USB. No rsync. No email to yourself.}
            \end{block}
            \vspace{0.5em}
            \begin{block}{\emph{Freedom to break things} $\triangleright$}
                Create an isolated copy of your work.\\
                Try something crazy. Throw it away if it fails.\\[4pt]
                \textit{Your main code is untouched.}
            \end{block}
    \end{columns}
\end{frame}

\section{Remote git}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{Connecting to a hub}
    \begin{columns}
        \column{0.52\textwidth}
            \uncover<1->{
            \begin{block}{Local repo exists $\triangleright$}
                Tell git where the remote is --- done \textbf{once}.\\[4pt]
                \texttt{git remote add origin}\\
                \texttt{\quad git@github.com:you/myproject.git}\\[4pt]
                \texttt{origin} is just a name --- an alias.\\
                You can have multiple remotes.
            \end{block}}
            \vspace{0.4em}
            \uncover<2->{
            \begin{block}{No local repo yet $\triangleright$}
                \texttt{git clone} creates the directory,\\
                initialises git, and connects the remote\\
                --- all in one step.\\[4pt]
                \texttt{git clone}\\
                \texttt{\quad git@github.com:you/myproject.git}
            \end{block}}
        \column{0.44\textwidth}
            \uncover<3->{
            \begin{block}{Directory vs Repository $\triangleright$}
                A \textbf{directory} is any folder\\
                on your filesystem.\\[6pt]
                A \textbf{repository} is a directory\\
                that git is tracking --- it has\\
                a \texttt{.git/} folder inside.\\[6pt]
                From the outside, they look identical.
            \end{block}}
    \end{columns}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{Sending and receiving}
    \begin{columns}
        \column{0.48\textwidth}
            \begin{block}{\texttt{git push} $\triangleright$}
                Send your local commits to the remote.\\[4pt]
                \begin{lstlisting}[language=bash]
git push
                \end{lstlisting}
                \textit{Your work is now backed up.\\
                Timestamped. Publicly yours.}
            \end{block}
        \column{0.48\textwidth}
            \pause
            \begin{block}{\texttt{git pull} $\triangleright$}
                Get the latest commits from the remote.\\[4pt]
                \begin{lstlisting}[language=bash]
git pull
                \end{lstlisting}
                \textit{Office, home, cluster.\\
                Always in sync.}
            \end{block}
    \end{columns}
    \vspace{0.8em}
    \pause
    \centering
    \texttt{[Your laptop \;\; .git/]}
    \quad
    $\begin{array}{c}
        \xrightarrow{\quad\texttt{push}\quad}\\[-4pt]
        \xleftarrow{\quad\texttt{pull}\quad}
    \end{array}$
    \quad
    \texttt{[GitHub]}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Why push every day?}
    \begin{columns}
        \column{0.5\textwidth}
            \begin{alertblock}{Local commits die with your laptop}
                A commit saved only locally disappears if:
                \begin{itemize}
                    \item Hard drive fails
                    \item Laptop is stolen
                    \item Accidental \texttt{rm -rf}
                \end{itemize}
            \end{alertblock}
        \column{0.5\textwidth}
            \pause
            \begin{block}{A push is different $\triangleright$}
                \begin{itemize}
                    \item Your work lives on a server
                    \item Timestamped --- publicly
                    \item \textbf{Nobody can dispute\\who wrote it first}
                    \item One push costs less data\\than sending one email
                \end{itemize}
            \end{block}
    \end{columns}
    \vspace{1em}
    \pause
    \centering
    \textbf{Push once or twice a day.\\
    No bandwidth concern. No data concern.\\
    There is only upside.}
\end{frame}

% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}[fragile]{The daily habit}
    \centering
    \begin{block}{Morning --- get what changed $\triangleright$}
        \begin{lstlisting}[language=bash]
git pull
        \end{lstlisting}
    \end{block}
    \vspace{0.6em}
    \begin{block}{Evening --- save your day $\triangleright$}

git add -u

git push
        \end{lstlisting}
    \end{block}
    \vspace{0.8em}
    \pause
    \textbf{Three commands. \; Thirty seconds. \; Every day.}\\[6pt]
    \small Your laptop can burn tonight. \; Your work is safe.
\end{frame}

\begin{frame}[fragile]{Summary}
